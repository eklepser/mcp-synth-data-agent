from __future__ import annotations

import warnings
from dataclasses import dataclass

import numpy as np
import pandas as pd
from copulas.multivariate import GaussianMultivariate


@dataclass
class CopulaModel:
    columns: list[str]
    model: GaussianMultivariate | None = None
    random_state: int = 42

    def fit(self, df: pd.DataFrame) -> "CopulaModel":
        numeric = df[self.columns].dropna()
        if numeric.empty:
            raise ValueError("No complete rows available for copula training")
        np.random.seed(self.random_state)
        self.model = GaussianMultivariate()
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            self.model.fit(numeric)
        return self

    def sample(self, n: int) -> pd.DataFrame:
        if self.model is None:
            raise RuntimeError("Model is not fitted - call fit() first")
        np.random.seed(self.random_state)
        synth = self.model.sample(n)
        return synth[self.columns].reset_index(drop=True)


def fit_copula(df: pd.DataFrame, columns: list[str], random_state: int = 42) -> CopulaModel:
    return CopulaModel(columns=columns, random_state=random_state).fit(df)

def apply_typo(value: str, rng: random.Random) -> str:
    if not isinstance(value, str) or len(value) < 2:
        return value
    idx = rng.randint(0, len(value) - 2)
    chars = list(value)
    chars[idx], chars[idx + 1] = chars[idx + 1], chars[idx]
    return "".join(chars)

def apply_numeric_typo(value: float, rng: random.Random) -> float:
    kind = rng.choice(["shift", "drop_zero", "duplicate_digit"])
    if kind == "shift":
        return value * rng.choice([0.1, 10.0])
    elif kind == "drop_zero":
        s = str(value)
        return float(s.replace("0", "", 1)) if "0" in s else value
    else:
        s = str(value)
        if len(s) < 2:
            return value
        idx = rng.randint(0, len(s) - 1)
        return float(s[:idx] + s[idx] + s[idx:])

def apply_outlier(value: float, rng: random.Random) -> float:
    multiplier = rng.choice([0.01, 0.1, 10.0, 100.0, -1.0])
    return float(value * multiplier)

def apply_missing(_: Any, __: random.Random) -> float:
    return np.nan

def apply_category_swap(value: Any, rng: random.Random, valid_categories: list) -> Any:
    alternatives = [v for v in valid_categories if v != value]
    if not alternatives:
        return value
    return rng.choice(alternatives)

ERROR_TYPE_FUNCS = {
    "typo": apply_typo,
    "numeric_typo": apply_numeric_typo,
    "outlier": apply_outlier,
    "missing": apply_missing,
    "category_swap": apply_category_swap,
}

def get_ecar_mask(n: int, error_rate: float, rng: np.random.Generator) -> np.ndarray:
    return rng.random(n) < error_rate

def get_enar_mask(series: pd.Series, error_rate: float, condition: str, rng: np.random.Generator) -> np.ndarray:
    try:
        condition_mask = series.to_frame().eval(condition)
    except Exception as e:
        raise ValueError(f"Не удалось оценить условие '{condition}': {e}")
    base_mask = rng.random(len(series)) < error_rate
    return base_mask & condition_mask.values

@dataclass
class ColumnErrorConfig:
    error_type: Literal["typo", "numeric_typo", "outlier", "missing", "category_swap"]
    error_rate: float
    mechanism: Literal["ecar", "enar"] = "ecar"
    condition: str | None = None

@dataclass
class PairwiseDistortionConfig:
    """Конфигурация для парных искажений (ТЕЗИС: разрыв корреляции, инверсия)"""
    col1: str
    col2: str
    action: Literal["break_correlation", "invert_correlation"]
    target_correlation: float = 0.0  # Для break_correlation цель ~0.0

@dataclass
class DistortionReport:
    """Структурированный отчет о примененных искажениях (ТЕЗИС: возвращая структурированный отчёт)"""
    target_error_rate: float
    actual_error_rate: float
    corrupted_cells_count: int
    applied_marginal_models: list[dict] = field(default_factory=list)
    applied_pairwise_actions: list[dict] = field(default_factory=list)
    structural_changes: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return asdict(self)

@dataclass
class ErrorModel:
    column_configs: dict[str, ColumnErrorConfig] = field(default_factory=dict)
    pairwise_configs: list[PairwiseDistortionConfig] = field(default_factory=list)

    def to_dict(self) -> dict:
        result = {"marginal": {}, "pairwise": []}
        for col, cfg in self.column_configs.items():
            result["marginal"][col] = asdict(cfg)
        for cfg in self.pairwise_configs:
            result["pairwise"].append(asdict(cfg))
        return result

def load_error_model(json_path: str) -> ErrorModel:
    with open(json_path, "r", encoding="utf-8") as f:
        data = json.load(f)
    
    configs = {}
    for col, cfg_dict in data.get("marginal", {}).items():
        configs[col] = ColumnErrorConfig(**cfg_dict)
    
    pairwise = [PairwiseDistortionConfig(**p) for p in data.get("pairwise", [])]
    return ErrorModel(column_configs=configs, pairwise_configs=pairwise)

def inject_errors(
    df: pd.DataFrame,
    model: ErrorModel,
    random_state: int = 42,
) -> tuple[pd.DataFrame, pd.DataFrame, DistortionReport]:
    py_rng = random.Random(random_state)
    np_rng = np.random.default_rng(random_state)

    corrupted_df = df.copy()
    error_mask = pd.DataFrame(False, index=df.index, columns=df.columns)
    total_cells = len(df) * len(df.columns)
    corrupted_count = 0
    applied_marginal = []

    # 1. Маргинальные искажения
    for col, config in model.column_configs.items():
        if col not in corrupted_df.columns:
            continue

        series = corrupted_df[col]
        non_null_mask = series.notna()

        if config.mechanism == "enar" and config.condition:
            mask = get_enar_mask(series, config.error_rate, config.condition, np_rng)
            mask = mask & non_null_mask.values
        else:
            mask = get_ecar_mask(len(series), config.error_rate, np_rng)
            mask = mask & non_null_mask.values

        error_mask.loc[mask, col] = True
        corrupted_count += int(mask.sum())
        applied_marginal.append({"column": col, "type": config.error_type, "rate": config.error_rate, "corrupted": int(mask.sum())})

        error_func = ERROR_TYPE_FUNCS.get(config.error_type)
        if error_func is None:
            raise ValueError(f"Неизвестный тип ошибки: {config.error_type}")

        for idx in series[mask].index:
            val = series.loc[idx]
            if config.error_type == "category_swap":
                valid_cats = series.dropna().unique().tolist()
                corrupted_df.loc[idx, col] = error_func(val, py_rng, valid_cats)
            else:
                corrupted_df.loc[idx, col] = error_func(val, py_rng)

    # 2. Парные и структурные искажения (ТЕЗИС: разрыв корреляции и др.)
    applied_pairwise = []
    structural_changes = {}

    for p_cfg in model.pairwise_configs:
        if p_cfg.col1 not in corrupted_df.columns or p_cfg.col2 not in corrupted_df.columns:
            continue
        
        # Вычисляем исходную корреляцию (только для числовых)
        try:
            orig_corr = float(corrupted_df[[p_cfg.col1, p_cfg.col2]].corr().iloc[0, 1])
        except Exception:
            orig_corr = None

        if p_cfg.action == "break_correlation":
            # Перемешиваем col2, сохраняя маргинальное распределение, но разрывая связь
            shuffled = corrupted_df[p_cfg.col2].sample(frac=1.0, random_state=np_rng).reset_index(drop=True)
            corrupted_df[p_cfg.col2] = shuffled
            
            try:
                new_corr = float(corrupted_df[[p_cfg.col1, p_cfg.col2]].corr().iloc[0, 1])
            except Exception:
                new_corr = None
                
            applied_pairwise.append({
                "action": "break_correlation",
                "col1": p_cfg.col1,
                "col2": p_cfg.col2,
                "original_correlation": orig_corr,
                "new_correlation": new_corr,
                "target_correlation": p_cfg.target_correlation
            })
        elif p_cfg.action == "invert_correlation":
            # Инверсия знака (для числовых)
            if pd.api.types.is_numeric_dtype(corrupted_df[p_cfg.col2]):
                mean_val = corrupted_df[p_cfg.col2].mean()
                corrupted_df[p_cfg.col2] = mean_val - (corrupted_df[p_cfg.col2] - mean_val)
                applied_pairwise.append({"action": "invert_correlation", "col1": p_cfg.col1, "col2": p_cfg.col2})

    actual_error_rate = corrupted_count / total_cells if total_cells > 0 else 0.0

    report = DistortionReport(
        target_error_rate=sum(c.error_rate for c in model.column_configs.values()) / max(1, len(model.column_configs)),
        actual_error_rate=float(actual_error_rate),
        corrupted_cells_count=corrupted_count,
        applied_marginal_models=applied_marginal,
        applied_pairwise_actions=applied_pairwise,
        structural_changes=structural_changes
    )

    return corrupted_df, error_mask, report

def auto_error_model_from_profile(
    profile: TableProfile,
    error_rate: float = 0.1,
    exclude_columns: list[str] | None = None,
) -> ErrorModel:
    exclude_columns = exclude_columns or []
    configs = {}

    for col_profile in profile.columns:
        col_name = col_profile.name
        if col_name in exclude_columns or col_profile.missing_fraction > 0.5:
            continue

        kind = col_profile.kind
        if kind == "numeric":
            configs[col_name] = ColumnErrorConfig(error_type="outlier", error_rate=error_rate, mechanism="ecar")
        elif kind == "categorical" and len(col_profile.categories) > 1:
            configs[col_name] = ColumnErrorConfig(error_type="category_swap", error_rate=error_rate, mechanism="ecar")
        elif kind == "boolean" and len(col_profile.bool_values) > 1:
            configs[col_name] = ColumnErrorConfig(error_type="category_swap", error_rate=error_rate, mechanism="ecar")
        elif kind == "text":
            configs[col_name] = ColumnErrorConfig(error_type="typo", error_rate=error_rate, mechanism="ecar")
        elif kind == "datetime":
            configs[col_name] = ColumnErrorConfig(error_type="typo", error_rate=error_rate, mechanism="ecar")

    return ErrorModel(column_configs=configs)

def save_error_model(model: ErrorModel, json_path: str) -> None:
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(model.to_dict(), f, indent=2, ensure_ascii=False)

from __future__ import annotations

import random
import string
import uuid
import numpy as np
import pandas as pd

def generate_numeric(df: pd.DataFrame, profile: TableProfile, n: int, random_state: int = 42, clip_to_range: bool = True) -> pd.DataFrame:
    numeric_cols = [c.name for c in profile.columns if c.kind == "numeric"]
    if not numeric_cols:
        return pd.DataFrame()

    copula = fit_copula(df, numeric_cols, random_state=random_state)
    synth = copula.sample(n)
    rng = np.random.default_rng(random_state)

    for c in profile.columns:
        if c.kind != "numeric":
            continue
            
        if c.is_integer:
            rounded = np.round(synth[c.name].to_numpy())
            if c.min is not None: 
                rounded = np.maximum(rounded, np.floor(c.min))
            if c.max is not None: 
                rounded = np.minimum(rounded, np.ceil(c.max))
            synth[c.name] = rounded.astype("int64")
        else:
            vals = synth[c.name].to_numpy()
            
            if c.min is not None and c.max is not None:
                cur_min, cur_max = np.min(vals), np.max(vals)
                if cur_max > cur_min:
                    vals = c.min + (vals - cur_min) * (c.max - c.min) / (cur_max - cur_min)
                else:
                    vals = np.full_like(vals, c.min)
                
                vals = np.clip(vals, c.min, c.max)

            if c.std is not None and c.std > 0:
                cur_std = np.std(vals)
                if cur_std > 0:
                    target_mean = c.mean if c.mean is not None else np.mean(vals)
                    vals = target_mean + (vals - np.mean(vals)) * (c.std / cur_std)
                    if c.min is not None and c.max is not None:
                        vals = np.clip(vals, c.min, c.max)

            span = (c.max - c.min) if (c.min is not None and c.max is not None) else 1.0
            delta = max(1e-5, span * 0.0001)
            vals = vals + rng.uniform(-delta, delta, size=len(vals))
            
            synth[c.name] = vals

    return synth


def generate_categorical(profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(random_state)
    result = {}
    for c in profile.columns:
        if c.kind == "categorical" and c.categories:
            cats = c.categories
            probs = np.array([c.category_freqs[str(k)] for k in cats])
            probs = probs / probs.sum()
            result[c.name] = rng.choice(cats, size=n, p=probs)
    return pd.DataFrame(result)


def generate_boolean(profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(random_state)
    result = {}
    for c in profile.columns:
        if c.kind == "boolean" and c.bool_values:
            vals = c.bool_values
            probs = np.array([c.bool_freqs[str(k)] for k in vals])
            probs = probs / probs.sum()
            result[c.name] = rng.choice(vals, size=n, p=probs)
    return pd.DataFrame(result)


def generate_identifier(df: pd.DataFrame, profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(random_state)
    result = {}
    for c in profile.columns:
        if c.kind == "identifier" and c.id_is_integer and c.id_min is not None:
            start = int(round(c.id_min))
            vals = np.arange(start, start + n, dtype=np.int64)
            rng.shuffle(vals)
            result[c.name] = vals
    return pd.DataFrame(result)


def generate_datetime(profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    rng = np.random.default_rng(random_state)
    result = {}
    for c in profile.columns:
        if c.kind == "datetime" and c.min is not None and c.max is not None:
            lo, hi = int(c.min), int(c.max)
            if hi <= lo:
                result[c.name] = np.full(n, lo, dtype="int64")
                continue
            offsets = rng.integers(lo, hi, size=n, dtype="int64")
            ts = pd.to_datetime(offsets, unit="ns")
            result[c.name] = ts.strftime(c.datetime_format) if c.datetime_format else ts
    return pd.DataFrame(result)


def generate_text(profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    py_rng = random.Random(random_state)
    result = {}
    for c in profile.columns:
        if c.kind != "text":
            continue

        if c.text_pattern == "uuid":
            generated = set()
            while len(generated) < n:
                generated.add(str(uuid.UUID(int=py_rng.getrandbits(128))))
            result[c.name] = list(generated)

        elif c.text_pattern == "mac":
            generated = set()
            while len(generated) < n:
                generated.add(":".join(f"{py_rng.randint(0, 255):02X}" for _ in range(6)))
            result[c.name] = list(generated)

        elif c.text_pattern == "phone":
            length = c.text_options.get("length", 12)
            generated = set()
            while len(generated) < n:
                prefix = "+7" 
                rest_len = length - 2
                rest = "".join(str(py_rng.randint(0, 9)) for _ in range(rest_len))
                generated.add(f"{prefix}{rest}")
            result[c.name] = list(generated)

        elif c.text_pattern == "pattern":
            pattern = c.text_options["pattern"]
            generated = set()
            max_attempts = n * 10 
            attempts = 0
            
            while len(generated) < n and attempts < max_attempts:
                attempts += 1
                s_parts = []
                for item in pattern:
                    item_type = item[0]
                    if item_type == "literal":
                        s_parts.append(item[1])
                    elif item_type == "number":
                        min_val, max_val, min_len, max_len = item[1], item[2], item[3], item[4]
                        val = py_rng.randint(min_val, max_val)
                        s_parts.append(f"{val:0{max_len}d}" if len(str(val)) < min_len else str(val))
                    elif item_type == "word":
                        words, is_lower, is_upper = item[1], item[2], item[3]
                        word = py_rng.choice(words)
                        if is_lower: word = word.lower()
                        elif is_upper: word = word.upper()
                        s_parts.append(word)
                    else:
                        min_len, max_len = item[1], item[2]
                        length = py_rng.randint(min_len, max_len)
                        s_parts.append("".join(py_rng.choice(string.ascii_letters) for _ in range(length)))
                generated.add("".join(s_parts))
            
            result[c.name] = list(generated)
            while len(result[c.name]) < n:
                result[c.name].append(py_rng.choice(result[c.name]))
                
        else:
            generated = set()
            while len(generated) < n:
                generated.add("".join(py_rng.choices(string.ascii_lowercase, k=8)))
            result[c.name] = list(generated)

    return pd.DataFrame(result)


def generate_table(df: pd.DataFrame, profile: TableProfile, n: int, random_state: int = 42) -> pd.DataFrame:
    parts = []
    parts.append(generate_numeric(df, profile, n, random_state))
    parts.append(generate_categorical(profile, n, random_state))
    parts.append(generate_boolean(profile, n, random_state))
    parts.append(generate_identifier(df, profile, n, random_state))
    parts.append(generate_datetime(profile, n, random_state))
    parts.append(generate_text(profile, n, random_state))

    synth = pd.concat([p for p in parts if not p.empty], axis=1)

    rng = np.random.default_rng(random_state + 1)
    for c in profile.columns:
        if c.name in synth.columns and c.missing_fraction > 0:
            mask = rng.random(n) < c.missing_fraction
            synth.loc[mask, c.name] = np.nan

    ordered = [c.name for c in profile.columns if c.name in synth.columns]
    return synth[ordered].reset_index(drop=True)

def ks_metric(real: pd.Series, synth: pd.Series) -> dict[str, float]:
    r = real.dropna().to_numpy()
    s = synth.dropna().to_numpy()
    if len(r) == 0 or len(s) == 0:
        return {"ks": float("nan"), "p_value": float("nan")}
    result = stats.ks_2samp(r, s)
    return {"ks": float(result.statistic), "p_value": float(result.pvalue)}

def tv_metric(real: pd.Series, synth: pd.Series) -> float:
    r = real.dropna().astype(str).value_counts(normalize=True)
    s = synth.dropna().astype(str).value_counts(normalize=True)
    cats = sorted(set(r.index) | set(s.index))
    p = np.array([r.get(c, 0.0) for c in cats])
    q = np.array([s.get(c, 0.0) for c in cats])
    return float(0.5 * np.abs(p - q).sum())

def frobenius_corr_metric(real: pd.DataFrame, synth: pd.DataFrame, cols: list[str]) -> float:
    if not cols:
        return 0.0
    r = real[cols].corr().to_numpy()
    s = synth[cols].corr().to_numpy()
    return float(np.linalg.norm(r - s, "fro"))

def js_metric(real: pd.Series, synth: pd.Series, n_bins: int = 20) -> float:
    r = real.dropna().to_numpy()
    s = synth.dropna().to_numpy()
    if len(r) == 0 or len(s) == 0:
        return float("nan")
    lo = min(r.min(), s.min())
    hi = max(r.max(), s.max())
    if hi == lo:
        return 0.0
    bins = np.linspace(lo, hi, n_bins + 1)
    p, _ = np.histogram(r, bins=bins)
    q, _ = np.histogram(s, bins=bins)
    p = p / p.sum()
    q = q / q.sum()
    return float(jensenshannon(p, q) ** 2)

def _to_int64_ns(s: pd.Series) -> pd.Series:
    parsed = pd.to_datetime(s, errors="coerce")
    ns = parsed.astype("int64").astype("float64")
    ns = ns.where(parsed.notna(), other=np.nan)
    return ns

def _len_histogram(s: pd.Series, bins: int = 20) -> np.ndarray:
    lengths = s.dropna().astype(str).str.len().to_numpy()
    if lengths.size == 0:
        return np.zeros(bins)
    lo, hi = lengths.min(), lengths.max()
    if hi == lo:
        hist = np.zeros(bins)
        hist[0] = lengths.size
        return hist
    hist, _ = np.histogram(lengths, bins=np.linspace(lo, hi, bins + 1))
    return hist

def _check_identifier(real: pd.Series, synth: pd.Series, name: str) -> IdentifierCheck:
    s = synth.dropna()
    n_unique = int(s.nunique())
    unique_ok = n_unique == len(s)
    range_ok = True
    if pd.api.types.is_numeric_dtype(real) and pd.api.types.is_numeric_dtype(s):
        if len(s) == len(real):
            r_min, r_max = real.min(), real.max()
            s_min, s_max = s.min(), s.max()
            range_ok = bool(s_min >= r_min and s_max <= r_max)
    return IdentifierCheck(name=name, unique_ok=unique_ok, range_ok=range_ok, n_unique=n_unique)

def _check_text(real: pd.Series, synth: pd.Series, name: str) -> TextCheck:
    r = real.dropna().astype(str)
    s = synth.dropna().astype(str)
    real_mean = float(r.str.len().mean()) if len(r) else float("nan")
    synth_mean = float(s.str.len().mean()) if len(s) else float("nan")
    r_set = set(r.tolist())
    s_set = set(s.tolist())
    union = r_set | s_set
    overlap = len(r_set & s_set) / len(union) if union else 0.0
    return TextCheck(
        name=name, real_mean_len=real_mean, synth_mean_len=synth_mean,
        real_unique=len(r_set), synth_unique=len(s_set), overlap_share=float(overlap),
    )

def evaluate_reproduction(
    real: pd.DataFrame,
    synth: pd.DataFrame,
    profile=None,
) -> ReproductionReport:
    common = [c for c in real.columns if c in synth.columns]
    report = ReproductionReport(n_rows_real=len(real), n_rows_synth=len(synth))

    if profile is not None:
        kinds = {c.name: c.kind for c in profile.columns}
    else:
        kinds = {}
        for c in common:
            if pd.api.types.is_numeric_dtype(real[c]):
                kinds[c] = "numeric"
            else:
                kinds[c] = "categorical"

    numeric_used = []
    for name in common:
        kind = kinds.get(name, "categorical")
        cm = ColumnMetrics(name=name, kind=kind)
        if kind == "numeric":
            ks = ks_metric(real[name], synth[name])
            cm.ks = ks["ks"]
            cm.ks_p_value = ks["p_value"]
            cm.js = js_metric(real[name], synth[name])
            numeric_used.append(name)
        elif kind in ("categorical", "boolean"):
            cm.tv = tv_metric(real[name], synth[name])
        elif kind == "identifier":
            report.identifier_checks.append(_check_identifier(real[name], synth[name], name))
        elif kind == "datetime":
            real_ns = _to_int64_ns(real[name])
            synth_ns = _to_int64_ns(synth[name])
            ks = ks_metric(real_ns, synth_ns)
            cm.ks = ks["ks"]
            cm.ks_p_value = ks["p_value"]
            cm.js = js_metric(real_ns, synth_ns)
        elif kind == "text":
            report.text_checks.append(_check_text(real[name], synth[name], name))
            cm.js = js_metric(
                pd.Series(_len_histogram(real[name])),
                pd.Series(_len_histogram(synth[name])),
            )
        report.columns.append(cm)

    report.numeric_columns_used = numeric_used
    report.frobenius_corr = frobenius_corr_metric(real, synth, numeric_used)
    return report

@dataclass
class DistortionEvaluationReport:
    frobenius_corr_shift: float  # Норма Фробениуса разности корреляционных матриц
    variance_growth_avg: float   # Средний относительный прирост дисперсии
    pairwise_targets_met: list[dict] # Выполнение целей по парным искажениям

    def to_dict(self) -> dict[str, Any]:
        return asdict(self)

def evaluate_distortion(
    original_df: pd.DataFrame,
    corrupted_df: pd.DataFrame,
    distortion_report: DistortionReport,
) -> DistortionEvaluationReport:
    numeric_cols = [c for c in original_df.columns if pd.api.types.is_numeric_dtype(original_df[c]) and pd.api.types.is_numeric_dtype(corrupted_df[c])]
    
    # 1. Сдвиг корреляционной матрицы (Норма Фробениуса)
    frob_shift = frobenius_corr_metric(original_df, corrupted_df, numeric_cols)
    
    # 2. Прирост дисперсии
    var_growth_list = []
    for col in numeric_cols:
        var_orig = original_df[col].var()
        var_corr = corrupted_df[col].var()
        if var_orig > 1e-9:
            growth = (var_corr - var_orig) / var_orig
            var_growth_list.append(float(growth))
    
    avg_var_growth = float(np.mean(var_growth_list)) if var_growth_list else 0.0

    # 3. Проверка выполнения целей по парным искажениям
    pairwise_met = []
    for action in distortion_report.applied_pairwise_actions:
        if action["action"] == "break_correlation":
            target = action.get("target_correlation", 0.0)
            new_corr = action.get("new_correlation")
            if new_corr is not None:
                success = abs(new_corr - target) < 0.15  # Допуск 0.15
                pairwise_met.append({
                    "col1": action["col1"],
                    "col2": action["col2"],
                    "target": target,
                    "achieved": new_corr,
                    "success": success
                })

    return DistortionEvaluationReport(
        frobenius_corr_shift=float(frob_shift),
        variance_growth_avg=avg_var_growth,
        pairwise_targets_met=pairwise_met
    )

def _detect_datetime_format(s: pd.Series) -> str | None:
    non_null = s.dropna()
    if len(non_null) == 0:
        return None
    sample = non_null.astype(str).head(50)
    for fmt in _DATETIME_FORMATS:
        try:
            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                parsed = pd.to_datetime(sample, format=fmt, errors="raise")
            if parsed.notna().all():
                return fmt
        except (ValueError, TypeError):
            continue
    return None

def _is_datetime_like(s: pd.Series) -> bool:
    if pd.api.types.is_datetime64_any_dtype(s):
        return True
    if not pd.api.types.is_object_dtype(s):
        return False
    non_null = s.dropna()
    if len(non_null) == 0:
        return False
    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            parsed = pd.to_datetime(non_null, errors="raise")
        return bool(
            parsed.notna().mean() > 0.99
            and non_null.astype(str).str.contains(r"[-:/]").mean() > 0.5
        )
    except (ValueError, TypeError):
        return False

def _is_dense_integer_identifier(s: pd.Series) -> bool:
    if not pd.api.types.is_numeric_dtype(s) or pd.api.types.is_bool_dtype(s):
        return False
    clean = s.dropna()
    if len(clean) == 0:
        return False
    vals = clean.astype(float).to_numpy()
    if not np.all(np.equal(np.mod(vals, 1), 0)):
        return False
    n_unique = clean.nunique()
    if n_unique / len(clean) < _IDENTIFIER_UNIQUE_THRESHOLD:
        return False
    span = float(np.max(vals) - np.min(vals))
    if span == 0:
        return False
    return (span / len(clean)) < _IDENTIFIER_DENSITY_THRESHOLD

def _is_integer_column(s: pd.Series) -> bool:
    if not pd.api.types.is_numeric_dtype(s) or pd.api.types.is_bool_dtype(s):
        return False
    clean = s.dropna()
    if len(clean) == 0:
        return False
    vals = clean.astype(float).to_numpy()
    is_int = np.equal(np.mod(vals, 1), 0)
    return bool(is_int.mean() >= _INTEGER_SHARE_THRESHOLD)

def _looks_like_text(s: pd.Series) -> bool:
    if not pd.api.types.is_object_dtype(s):
        return False
    clean = s.dropna().astype(str)
    if len(clean) == 0:
        return False
    if clean.nunique() / len(clean) < _TEXT_UNIQUE_THRESHOLD:
        return False
    return bool(clean.str.len().mean() >= _TEXT_MIN_MEAN_LEN)

def _classify_series(s: pd.Series) -> str:
    if _is_datetime_like(s):
        return "datetime"

    non_null = s.dropna()
    n_nonnull = len(non_null)
    if n_nonnull == 0:
        return "categorical"

    n_unique = non_null.nunique()
    if 0 < n_unique <= 2:
        return "boolean"

    if _is_dense_integer_identifier(s):
        return "identifier"

    if pd.api.types.is_numeric_dtype(s) and not pd.api.types.is_bool_dtype(s):
        return "numeric"

    if _looks_like_text(s):
        return "text"

    return "categorical"

def _dist_arg_names(dist) -> list[str]:
    shapes = list(getattr(dist, "shapes", "") or "")
    if isinstance(shapes, str):
        shapes = [s.strip() for s in shapes.split(",") if s.strip()]
    return shapes + ["loc", "scale"]

def _fit_numeric(values: np.ndarray) -> tuple[str, dict[str, float], float]:
    best_name, best_params, best_aic = "norm", {}, float("inf")
    for name, dist in _NUMERIC_CANDIDATES:
        try:
            with warnings.catch_warnings():
                warnings.simplefilter("ignore")
                params = dist.fit(values)
                loglik = np.sum(dist.logpdf(values, *params))
            if not np.isfinite(loglik):
                continue
            k = len(params)
            aic = 2 * k - 2 * loglik
            if aic < best_aic:
                best_name, best_params, best_aic = name, params, aic
        except Exception:
            continue
    if best_aic == float("inf"):
        mu, sigma = float(np.mean(values)), float(np.std(values) or 1.0)
        best_name, best_params, best_aic = "norm", (mu, sigma), float("inf")
    dist = dict(_NUMERIC_CANDIDATES)[best_name]
    arg_names = _dist_arg_names(dist)
    params_dict = {n: float(p) for n, p in zip(arg_names, best_params)}
    return best_name, params_dict, float(best_aic)

def _extract_text_pattern(strings: list[str]) -> tuple[str, dict[str, Any]]:
    if not strings:
        return "generic", {}
    
    # 1. UUID
    if all(re.match(r'^[0-9a-fA-F]{8}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{4}-[0-9a-fA-F]{12}$', s) for s in strings):
        return "uuid", {}

    # 2. MAC-адрес
    if all(re.match(r'^[0-9A-Fa-f]{2}(:[0-9A-Fa-f]{2}){5}$', s) for s in strings):
        return "mac", {}

    # 3. Телефон
    if all(x.startswith("+") for x in strings):
        return "phone", {"length": len(strings[0])}

    # 4. Универсальный токенизатор
    tokenized = [re.findall(r'[0-9]+|[a-zA-Z]+|.', s) for s in strings]
    lengths = [len(t) for t in tokenized]
    if not lengths:
        return "generic", {}
        
    most_common_len = Counter(lengths).most_common(1)[0][0]
    filtered = [t for t in tokenized if len(t) == most_common_len]
    
    pattern = []
    for i in range(most_common_len):
        tokens = [t[i] for t in filtered]
        
        if len(set(tokens)) == 1:
            pattern.append(("literal", tokens[0]))
            continue
            
        if all(t.isdigit() for t in tokens):
            nums = [int(t) for t in tokens]
            pattern.append(("number", min(nums), max(nums), min(len(t) for t in tokens), max(len(t) for t in tokens)))
            continue
            
        if all(t.isalpha() for t in tokens):
            unique_words = list(set(tokens)) 
            is_lower = all(t.islower() for t in tokens)
            is_upper = all(t.isupper() for t in tokens)
            pattern.append(("word", unique_words, is_lower, is_upper))
            continue
            
        pattern.append(("any", min(len(t) for t in tokens), max(len(t) for t in tokens)))

    return "pattern", {"pattern": pattern, "length": most_common_len}

def _parse_str_column(s: pd.Series) -> pd.Series:
    clean = s.dropna()
    if len(clean) == 0:
        return s

    try:
        sample = clean.iloc[0]
        if isinstance(sample, str):
            if sample.startswith("0") or any(c in sample for c in "+ :"):
                raise ValueError("Force string")
            if sample.isdigit() and len(sample) >= 5:
                raise ValueError("Force string (long digits)")
                
            _ = pd.to_numeric(clean, errors="raise")
            return pd.to_numeric(s, errors="raise")
    except (ValueError, TypeError, AttributeError):
        pass

    try:
        with warnings.catch_warnings():
            warnings.simplefilter("ignore")
            parsed = pd.to_datetime(clean, errors="raise")
        if parsed.dt.year.min() > 1900 and parsed.dt.year.max() < 2100:
            return pd.to_datetime(s, errors="coerce")
    except (ValueError, TypeError):
        pass

    return s

def profile_column(s: pd.Series) -> ColumnProfile:
    kind = _classify_series(s)
    n_total = len(s)
    n_missing = int(s.isna().sum())
    cp = ColumnProfile(
        name=str(s.name), kind=kind, n_total=n_total, n_missing=n_missing,
        missing_fraction=n_missing / n_total if n_total else 0.0,
    )

    if kind == "numeric":
        clean = s.dropna().astype(float).to_numpy()
        if clean.size == 0: return cp
        name, params, aic = _fit_numeric(clean)
        cp.distribution, cp.dist_params, cp.dist_aic = name, params, aic
        cp.mean = float(np.mean(clean))
        cp.std = float(np.std(clean, ddof=1)) if clean.size > 1 else 0.0
        cp.min, cp.max = float(np.min(clean)), float(np.max(clean))
        cp.q01 = float(np.quantile(clean, 0.01))
        cp.q99 = float(np.quantile(clean, 0.99))
        cp.is_integer = _is_integer_column(s)

    elif kind == "identifier":
        clean = s.dropna()
        vals = clean.astype(float).to_numpy()
        cp.id_min, cp.id_max = float(np.min(vals)), float(np.max(vals))
        cp.id_is_integer = True

    elif kind == "boolean":
        vc = s.dropna().value_counts(normalize=True)
        cp.bool_values = [vc.index[i].item() if hasattr(vc.index[i], "item") else vc.index[i] for i in range(len(vc))]
        cp.bool_freqs = {str(k): float(v) for k, v in vc.items()}

    elif kind == "categorical":
        clean = s.dropna().astype(str)
        vc = clean.value_counts(normalize=True)
        cp.categories = vc.index.tolist()
        cp.category_freqs = {str(k): float(v) for k, v in vc.items()}
        if len(clean) > 0:
            cp.high_cardinality = (clean.nunique() / len(clean)) >= _HIGH_CARDINALITY_THRESHOLD

    elif kind == "text":
        clean = s.dropna().astype(str).tolist()
        cp.text_pattern, cp.text_options = _extract_text_pattern(clean)

    elif kind == "datetime":
        cp.datetime_format = _detect_datetime_format(s)
        clean = pd.to_datetime(s.dropna(), errors="coerce").dropna()
        if len(clean):
            cp.min, cp.max = float(clean.astype("int64").min()), float(clean.astype("int64").max())
    return cp

def profile_table(df: pd.DataFrame) -> TableProfile:
    columns = [profile_column(df[c]) for c in df.columns]
    schema = {c.name: c.kind for c in columns}
    return TableProfile(n_rows=len(df), n_cols=df.shape[1], columns=columns, schema=schema)

def profile_csv(path: str) -> TableProfile:
    df_str = pd.read_csv(path, dtype=str)
    df_parsed = df_str.copy()
    for col in df_str.columns:
        df_parsed[col] = _parse_str_column(df_str[col])
    return profile_table(df_parsed)
